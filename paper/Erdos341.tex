\begin{abstract}
Erd\H{o}s problem~341 asks whether the consecutive gap sequence of the pair-sum-avoiding greedy extension of every finite seed set is eventually periodic. We give a negative answer by exhibiting the following counterexample
\[
A=\{1,2,3,5,7,13,22,27,28,32,36,40,47,48,52,63,71,77,81,89,97\}.
\]
The proof constructs an infinite set \(S\) satisfying the exact greedy recurrence \(t\in S\Longleftrightarrow t\notin S+S\) for \(t>97\). We show that \(S\) is the greedy extension of the \(21\)-element seed set \(A\), and use a scale-eight sumset controller embedded in three residue classes modulo \(49\) and a finite modular shield to prove that the consecutive gap sequence of \(S\) is not eventually periodic. Motivated by this counterexample, we further obtain an infinite family of counterexamples by a slight modification of the construction.
\end{abstract}

\section{Introduction and the main theorem}

\noindent\textit{AI-use disclosure.} This proof was developed with assistance from OpenAI's GPT-5.6 Sol. All arguments were checked and verified by the author.

Write \(\N=\{1,2,3,\ldots\}\) and \(\Nzero=\{0,1,2,\ldots\}\). For \(X,Y\subseteq\Nzero\) and \(c\in\Nzero\), put
\[
X+Y=\{x+y\mid x\in X,\ y\in Y\},
\qquad
c+X=\{c+x\mid x\in X\},
\qquad
\cpl{X}=\Nzero\setminus X.
\]
Equal summands are allowed in every sumset. For integers \(a\leq b\), write \(\ints{a}{b}=\{a,a+1,\ldots,b\}\). 

Let \(A=\{a_1<\cdots<a_k\}\subseteq\N\) be a nonempty finite set. For \(n\geq k\), define
\begin{equation}\label{eq:greedy-definition}
a_{n+1}
=\min\{t\in\N\mid t>a_n,\ t\notin A_n+A_n\},
\qquad
A_n=\{a_1,\ldots,a_n\}.
\end{equation}
Let \(d_m=a_{m+1}-a_m\) denote the consecutive gaps. Erd\H{o}s and Graham asked whether, for every finite initial set \(A\), the resulting gap sequence \((d_m)_{m\ge 1}\) is eventually periodic. The question appears in Erd\H{o}s and Graham~\cite[p.~53]{ErdosGraham1980} and is catalogued as Erd\H{o}s Problem~341 in Bloom's online database~\cite{Bloom341}. Related periodicity criteria are due to Calkin and Finch~\cite{CalkinFinch1996}, while Calkin and Erd\H{o}s~\cite{CalkinErdos1996} proved that a natural class of aperiodic sum-free sets is incomplete. We answer the arbitrary finite-seed question negatively.

\begin{theorem}\label{thm:main}
Let
\[
A=\{1,2,3,5,7,13,22,27,28,32,36,40,47,48,52,63,71,77,81,89,97\}.
\]
For the sequence generated from \(A\) by~\eqref{eq:greedy-definition},
\[
\forall K\geq1\ \forall h\geq1\ \exists m\geq K
\quad d_{m+h}\neq d_m.
\]
Thus its consecutive gap sequence is not eventually periodic.
\end{theorem}

\begin{lemma}[Exact recurrence implies the greedy rule]\label{lem:greedy}
Let \(S\subseteq\N\) be infinite, let \(N\in S\), and put \(A=S\cap\ints{1}{N}\). Suppose that
\begin{equation}\label{eq:exact-recurrence}
t\in S
\quad\Longleftrightarrow\quad
t\notin S+S
\qquad (t>N).
\end{equation}
Then the increasing enumeration of \(S\) is exactly the greedy extension of \(A\).
\end{lemma}

\begin{proof}
Scan the integers \(t>N\) in order. Inductively, immediately before testing \(t\), the selected set is \(S\cap\ints{1}{t-1}\). Since all elements of \(S\) are positive, every representation \(t=u+v\) with \(u,v\in S\) has \(u,v<t\), and hence
\[
t\in(S\cap\ints{1}{t-1})+(S\cap\ints{1}{t-1})
\quad\Longleftrightarrow\quad t\in S+S.
\]
Thus~\eqref{eq:exact-recurrence} makes the rule select exactly the elements of \(S\), completing the induction.
\end{proof}

\section{A scale-eight sumset controller}

We use the standard identity
\begin{equation}
\ints{a}{b}+\ints{c}{d}=\ints{a+c}{b+d}
\qquad (a\leq b,\ c\leq d).
\end{equation}

Define
\begin{align}
L&=\bigcup_{k\geq0}\ints{8^k}{4\cdot8^k-1} \\
Y_0&=\{0\}\cup
      \bigcup_{k\geq0}\ints{4\cdot8^k}{8\cdot8^k-1},
      \label{eq:Y0}\\
Y_1&=\ints{1}{3}\cup
      \bigcup_{k\geq0}\ints{16\cdot8^k-1}{32\cdot8^k-1},
      \label{eq:Y1}\\
Y_2&=\ints{0}{1}\cup\ints{7}{15}\cup
      \bigcup_{k\geq0}\ints{64\cdot8^k-1}{128\cdot8^k-1}.
      \label{eq:Y2}
\end{align}
At each scale, the corresponding blocks of \(L\) and \(Y_0\setminus\{0\}\) partition \(\ints{8^k}{8^{k+1}-1}\). Thus
\begin{equation}\label{eq:Y0-complement-L}
Y_0=\cpl{L}.
\end{equation}

\begin{lemma}[Controller identities]
The sets in~\eqref{eq:Y0}--\eqref{eq:Y2} satisfy
\begin{align}
Y_1&=\cpl{(Y_0+Y_0)},                                      \label{eq:gate01}\\
Y_2&=\cpl{(Y_1+Y_1)},                                      \label{eq:gate12}\\
Y_0&=\{0\}\cup\bigl(1+\cpl{(Y_2+Y_2)}\bigr).             \label{eq:gate20}
\end{align}
\end{lemma}

\begin{proof}
A direct interval calculation gives
\begin{align}
Y_0+Y_0
&=\{0\}\cup\bigcup_{k\geq0}
  \ints{4\cdot8^k}{16\cdot8^k-2}=\cpl{Y_1},
  \label{eq:Y0-square}\\
Y_1+Y_1
&=\ints{2}{6}\cup\bigcup_{k\geq0}
  \ints{16\cdot8^k}{64\cdot8^k-2}=\cpl{Y_2},
  \label{eq:Y1-square}\\
Y_2+Y_2
&=\ints{0}{2}\cup\ints{7}{30}\cup\bigcup_{k\geq0}
  \ints{64\cdot8^k-1}{256\cdot8^k-2}=L-1.
  \label{eq:Y2-square}
\end{align}
Equations~\eqref{eq:Y2-square} and~\eqref{eq:Y0-complement-L} yield~\eqref{eq:gate20}.
\end{proof}

\begin{lemma}\label{lem:Y0-aperiodic}
The membership indicator of \(Y_0\) is not ultimately periodic.
\end{lemma}

\begin{proof}
Suppose that there are \(N\geq0\) and \(h\geq1\) such that
\[
n\in Y_0\iff n+h\in Y_0
\qquad(n\geq N).
\]
Choose \(a=8^k>\max\{N,h\}\) and put \(m=4a-h\). Then \(N<a\leq m\leq4a-1\). Hence \(m\in L\), so \(m\notin Y_0\), whereas \(m+h=4a\in\ints{4a}{8a-1}\subseteq Y_0\). This contradicts the assumed periodicity.
\end{proof}

\section{A finite shield modulo 49}

All residues in this section lie in \(\Zfortynine\) and are represented in \(\{0,\ldots,48\}\). Put
\begin{equation}
P=\{7,14,28\},\qquad B=\{3,22,32,40,48\},\qquad
D=\{1,2,5,13,27,36,47\},\qquad Q=P\cup B.
\end{equation}

\begin{lemma}[Finite shield]\label{lem:shield}
In \(\Zfortynine\),
\begin{align}
(B+B)\cup(P+B)\cup(D+B)&=\Zfortynine\setminus Q, \label{eq:cover-mod}\\
Q\cap\bigl((B+B)\cup(P+B)\cup(D+B)\cup(D+P)\bigr)
&=\varnothing,                                         \label{eq:avoid-mod}\\
(P+P)\cap Q&=P.                                        \label{eq:PP-mod}
\end{align}
Moreover, the only ordered pairs in \(P\times P\) whose sum lies in \(Q\) are
\begin{equation}\label{eq:diagonal-pairs}
7+7=14,
\qquad
14+14=28,
\qquad
28+28=49+7.
\end{equation}
\end{lemma}

\begin{proof}
All assertions follow by direct computation in \(\Zfortynine\).
\end{proof}

We now define the infinite set
\begin{equation}
\begin{split}
S={}&D\cup\{49q+b\mid q\in\Nzero,\ b\in B\}\\
&\cup\{49q+7\mid q\in Y_0\}
 \cup\{49q+14\mid q\in Y_1\}
 \cup\{49q+28\mid q\in Y_2\}.
\end{split}
\end{equation}
Hence
\begin{equation}\label{eq:seed-prefix}
S\cap\ints{1}{97}=D\cup B\cup(49+B)\cup\{7,28,63,77\}=A.
\end{equation}

\begin{theorem}\label{thm:recurrence}
For every \(t>97\),
\begin{equation}\label{eq:main-recurrence}
t\in S
\quad\Longleftrightarrow\quad
t\notin S+S.
\end{equation}
\end{theorem}

\begin{proof}
Write \(t=49q+r\), with \(0\leq r<49\). Since \(t>97\), we have $q\geq2$.

First let \(r\in Q=B\cup P\). The six possible pair-sum types are
\[
D+D,\quad D+B,\quad D+P,\quad B+B,\quad B+P,\quad P+P.
\]
A \(D+D\) sum is at most \(94<t\). Equation~\eqref{eq:avoid-mod} excludes the next four types from \(Q\), while~\eqref{eq:PP-mod} and~\eqref{eq:diagonal-pairs} leave only the three diagonal \(P+P\) pairs. Hence
\begin{align*}
49q+14\in S+S&\iff q\in Y_0+Y_0,\\
49q+28\in S+S&\iff q\in Y_1+Y_1,\\
49q+7\in S+S&\iff q-1\in Y_2+Y_2.
\end{align*}
Indeed, the corresponding representations are
\begin{align*}
(49u+7)+(49v+7)&=49(u+v)+14,\\
(49u+14)+(49v+14)&=49(u+v)+28,\\
(49u+28)+(49v+28)&=49(u+v+1)+7.
\end{align*}
The exclusion of every other source type proves the converse implication in each equivalence. The controller identities~\eqref{eq:gate01}--\eqref{eq:gate20} therefore make every \(P\)-rail integer represented exactly when it is absent from \(S\). Every \(B\)-rail integer belongs to \(S\), and the same exhaustive source analysis shows that none is represented. Thus~\eqref{eq:main-recurrence} holds for \(r\in Q\).

Now let \(r\notin Q\). Such an integer \(t\) is not in \(S\): its residue excludes every infinite rail, and \(t>97\) excludes the finite set \(D\). By~\eqref{eq:cover-mod}, there exist \(b\in B\) and \(x\in B\cup D\cup P\) such that \(x+b\equiv r\pmod {49}\). Set
\[
e_x=x\quad(x\in B\cup D),
\qquad e_7=7,\qquad e_{14}=63=49+14,\qquad e_{28}=28.
\]
Every \(e_x\) belongs to \(S\). Write
\[
e_x=49u_x+x,
\qquad x+b=49c+r.
\]
Here \(u_x,c\in\{0,1\}\). Since \(q\geq2\), the quotient \(q-u_x-c\) is nonnegative, and
\[
t=e_x+\bigl(49(q-u_x-c)+b\bigr)\in S+S.
\]
The second summand is a positive element of a full \(B\)-rail. Both summands are positive and sum to \(t\), so each is smaller than \(t\). This completes the proof.
\end{proof}

By~\eqref{eq:seed-prefix}, Theorem~\ref{thm:recurrence}, and Lemma~\ref{lem:greedy}, the increasing enumeration of \(S\) is exactly the greedy extension of the seed $A$ in Theorem~\ref{thm:main}.

\section{Nonperiodicity of the gaps}

\begin{lemma}\label{lem:gaps-imply-membership}
Let \(T=\{t_1<t_2<\cdots\}\subseteq\N\) be infinite. If the consecutive gaps \(t_{m+1}-t_m\) are eventually periodic, then the membership indicator of \(T\) is ultimately periodic.
\end{lemma}

\begin{proof}
Suppose that the gaps have period \(h\geq1\) from index \(K\) onward. Let
\[
H=\sum_{j=0}^{h-1}(t_{K+j+1}-t_{K+j})>0.
\]
Every block of \(h\) consecutive gaps in the periodic tail has sum $H$, so \(t_{m+h}=t_m+H\) for \(m\geq K\). Since \(T\) is infinite, it is unbounded. Thus, given \(n\geq t_K\), there is a unique \(m\geq K\) such that \(t_m\leq n<t_{m+1}\), and
\[
t_{m+h}\leq n+H<t_{m+h+1},
\qquad
(n+H)-t_{m+h}=n-t_m.
\]
Thus \(n=t_m\) if and only if \(n+H=t_{m+h}\), which proves
\[
n\in T\iff n+H\in T
\qquad(n\geq t_K).
\]
This is ultimate periodicity of the membership indicator.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
The residue-\(7\) rail of \(S\) satisfies
\begin{equation}\label{eq:rail-seven}
49q+7\in S\quad\Longleftrightarrow\quad q\in Y_0
\qquad(q\in\Nzero).
\end{equation}
Suppose, for contradiction, that membership in \(S\) were ultimately periodic with period \(h\geq1\). Then \(49h\) would also be an ultimate period. For all sufficiently large \(q\), equation~\eqref{eq:rail-seven} would give
\[
q\in Y_0
\iff49q+7\in S
\iff49(q+h)+7\in S
\iff q+h\in Y_0,
\]
contradicting Lemma~\ref{lem:Y0-aperiodic}. Thus the membership indicator of \(S\) is not ultimately periodic. The contrapositive of Lemma~\ref{lem:gaps-imply-membership} now shows that the consecutive gaps in the increasing enumeration of \(S\) are not eventually periodic, which is the assertion of the theorem.
\end{proof}

\section{An infinite family of counterexamples}

We further give an infinite family of counterexamples by replacing the finite shield above with a parameterized family of finite shields.

Fix an integer \(p\geq18\), put \(M=7p\), and work in \(\ZM\), represented by \(\{0,\ldots,M-1\}\). Define
\begin{equation}\label{eq:parametric-PBQFD}
\begin{gathered}
P_p=\{p,2p,4p\},
\qquad
B_p=\{M-3,M-1,6p+3\},
\qquad Q_p=P_p\cup B_p,\\
F_p=Q_p\cup(Q_p-B_p)\cup(Q_p-P_p),
\qquad
D_p=\ZM\setminus F_p,
\end{gathered}
\end{equation}
where \(Q_p-B_p=\{q-b:q\in Q_p,\ b\in B_p\}\), and similarly for \(Q_p-P_p\). Set
\begin{equation}\label{eq:pseed-family}
A_p=D_p\cup B_p\cup(M+B_p)\cup\{p,4p,M+2p,M+4p\}.
\end{equation}

\begin{theorem}\label{thm:pmain}
For every integer \(p\geq18\), the seed \(A_p\) in~\eqref{eq:pseed-family} has \(7p-20\) elements. For the sequence generated from \(A_p\) by~\eqref{eq:greedy-definition},
\[
\forall K\geq1\ \forall h\geq1\ \exists m\geq K
\quad d_{m+h}\neq d_m.
\]
Thus its consecutive-gap sequence is not eventually periodic. In particular, the sets \(A_p\), \(p\geq18\), form an infinite family of counterexamples.
\end{theorem}

All residues in this section lie in \(\ZM\), with the notation of~\eqref{eq:parametric-PBQFD}.

\begin{lemma}[Uniform finite shield]\label{lem:pshield}
In $\ZM$, for every \(p\geq18\),
\begin{align}
D_p+B_p&=\ZM\setminus Q_p,                                      \label{eq:pcover-mod}\\
Q_p\cap\bigl((B_p+B_p)\cup(P_p+B_p)\cup(D_p+B_p)\cup(D_p+P_p)\bigr)
&=\varnothing,                                             \label{eq:pavoid-mod}\\
(P_p+P_p)\cap Q_p&=P_p.                                            \label{eq:pPP-mod}
\end{align}
Moreover, the only ordered pairs in \(P_p\times P_p\) whose sum lies in \(Q_p\) are
\begin{equation}\label{eq:pdiagonal-pairs}
p+p=2p,
\qquad
2p+2p=4p,
\qquad
4p+4p=M+p.
\end{equation}
\end{lemma}

\begin{proof}
Expanding the definition in~\eqref{eq:parametric-PBQFD} gives
\begin{equation}\label{eq:pF-explicit}
\begin{split}
F_p=\{&0,2,\ p-6,p-4,p,p+1,p+3,\ 2p-3,2p,2p+1,2p+3,\\
&3p-3,3p-1,3p,\ 4p,4p+1,4p+3,\\
&5p-3,5p-1,5p,5p+3,\\
&6p-3,6p-1,6p,6p+3,6p+4,6p+6,\ M-3,M-2,M-1\}.
\end{split}
\end{equation}
For \(p\geq18\), these are \(30\) distinct representatives in \(\{0,\ldots,M-1\}\). In particular, \(|D_p|=M-30\). The definition of \(F_p\) immediately gives
\begin{equation}\label{eq:pdirect-avoidance}
(D_p+B_p)\cap Q_p=\varnothing,
\qquad
(D_p+P_p)\cap Q_p=\varnothing.
\end{equation}

We prove the reverse inclusion in~\eqref{eq:pcover-mod}. Let \(r\in\ZM\), and put \(f=r+1\). If \(f\notin F_p\), then \(f\in D_p\) and
\[
r=f+(M-1)\in D_p+B_p.
\]
It remains to consider \(f\in F_p\). Put
\[
G_p=\{0,p+1,2p+1,4p+1,6p+4,M-2\},
\qquad
E_p=\{p-6,3p-3,5p-3,6p-3,M-3\}.
\]
If \(f\in G_p\), then direct subtraction shows that \(r=f-1\in Q_p\). If \(f\in F_p\setminus(G_p\cup E_p)\), comparison with~\eqref{eq:pF-explicit} shows that \(f+2\notin F_p\), and hence
\[
r=(f+2)+(M-3)\in D_p+B_p.
\]
For the five values of \(f\) in \(E_p\), the corresponding representations are
\begin{align*}
p-7&=(2p-10)+(6p+3),&
3p-4&=(4p-7)+(6p+3),\\
5p-4&=(6p-7)+(6p+3),&
6p-4&=(M-7)+(6p+3),\\
M-4&=(p-7)+(6p+3).
\end{align*}
Thus every residue outside \(Q_p\) belongs to \(D_p+B_p\), and~\eqref{eq:pdirect-avoidance} proves~\eqref{eq:pcover-mod}.

Direct calculation shows that \(B_p+B_p\) and \(P_p+B_p\) are disjoint from \(Q_p\). Together with~\eqref{eq:pdirect-avoidance}, this proves~\eqref{eq:pavoid-mod}. Finally, the three mixed sums of elements of \(P_p\) are \(3p,5p,6p\), all outside \(Q_p\), while the diagonal sums are those in~\eqref{eq:pdiagonal-pairs}. This proves~\eqref{eq:pPP-mod} and the ordered-pair assertion.
\end{proof}

We now define the infinite set
\begin{equation}
\begin{split}
S_p={}&D_p\cup\{Mq+b\mid q\in\Nzero,\ b\in B_p\}\\
&\cup\{Mq+p\mid q\in Y_0\}
 \cup\{Mq+2p\mid q\in Y_1\}
 \cup\{Mq+4p\mid q\in Y_2\}.
\end{split}
\end{equation}
The initial controller memberships give
\begin{equation}\label{eq:pseed-prefix}
S_p\cap\ints{1}{2M-1}
=D_p\cup B_p\cup(M+B_p)\cup\{p,4p,M+2p,M+4p\}=A_p.
\end{equation}
In particular, \(2M-1=M+(M-1)\in A_p\). The four sets in~\eqref{eq:pseed-prefix} are disjoint, and therefore
\begin{equation}\label{eq:pseed-size}
|A_p|=(M-30)+3+3+4=M-20=7p-20.
\end{equation}

\begin{theorem}\label{thm:precurrence}
For every \(p\geq18\) and every \(t>2M-1\),
\begin{equation}
t\in S_p
\quad\Longleftrightarrow\quad
t\notin S_p+S_p.
\end{equation}
\end{theorem}

\begin{proof}
Write \(t=Mq+r\), with \(0\leq r<M\). Since \(t>2M-1\), we have \(q\geq2\). The argument is the same as in the proof of Theorem~\ref{thm:recurrence}, with \(M,Q_p,P_p,B_p,D_p,S_p\) replacing \(49,Q,P,B,D,S\) respectively, and Lemma~\ref{lem:pshield} replacing Lemma~\ref{lem:shield}. 

The only change occurs when \(r\notin Q_p\). Such \(t\) is not in \(S_p\). By~\eqref{eq:pcover-mod}, there exist \(d\in D_p\) and \(b\in B_p\) such that
\[
d+b=Mc+r
\qquad(c\in\{0,1\}).
\]
Since \(q\geq2\), we have \(q-c\geq1\), and
\[
t=d+\bigl(M(q-c)+b\bigr)\in S_p+S_p.
\]
This completes the proof.
\end{proof}

By~\eqref{eq:pseed-prefix}, Theorem~\ref{thm:precurrence}, and Lemma~\ref{lem:greedy}, the increasing enumeration of \(S_p\) is exactly the greedy extension of \(A_p\).

\begin{proof}[Proof of Theorem~\ref{thm:pmain}]
Equation~\eqref{eq:pseed-size} proves the asserted cardinality. The proof of nonperiodicity is the same as in Theorem~\ref{thm:main}. Indeed, suppose, for contradiction, that membership in \(S_p\) were ultimately periodic with period \(h\geq1\). Then for all sufficiently large \(q\),
\[
q\in Y_0
\iff Mq+p\in S_p
\iff M(q+h)+p\in S_p
\iff q+h\in Y_0,
\]
contradicting Lemma~\ref{lem:Y0-aperiodic}. Lemma~\ref{lem:gaps-imply-membership} therefore shows that the consecutive gaps in the greedy extension of \(A_p\) are not eventually periodic.
\end{proof}
